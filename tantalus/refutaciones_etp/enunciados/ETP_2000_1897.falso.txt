class Magma (α : Type _) where
  op : α → α → α

infix:65 " ◇ " => Magma.op

def LeyA (G : Type _) [Magma G] : Prop :=
  ∀ x y z : G, x = (y ◇ (z ◇ z)) ◇ (z ◇ x)

def LeyB (G : Type _) [Magma G] : Prop :=
  ∀ x y : G, x = (y ◇ (x ◇ y)) ◇ (y ◇ x)

theorem falsa_2000_1897 : ∀ (G : Type) [Magma G], LeyA G → LeyB G
