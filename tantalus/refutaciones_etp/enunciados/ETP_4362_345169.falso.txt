class Magma (α : Type _) where
  op : α → α → α

infix:65 " ◇ " => Magma.op

def LeyA (G : Type _) [Magma G] : Prop :=
  ∀ x y z : G, x ◇ (y ◇ z) = y ◇ (x ◇ z)

def LeyB (G : Type _) [Magma G] : Prop :=
  ∀ x y z : G, x = (y ◇ ((x ◇ y) ◇ y)) ◇ (x ◇ (z ◇ y))

theorem falsa_4362_345169 : ∀ (G : Type) [Magma G], LeyA G → LeyB G
