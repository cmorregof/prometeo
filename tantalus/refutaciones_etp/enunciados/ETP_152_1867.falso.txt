class Magma (α : Type _) where
  op : α → α → α

infix:65 " ◇ " => Magma.op

def LeyA (G : Type _) [Magma G] : Prop :=
  ∀ x y : G, x = (x ◇ x) ◇ (x ◇ y)

def LeyB (G : Type _) [Magma G] : Prop :=
  ∀ x y z : G, x = (x ◇ (y ◇ z)) ◇ (x ◇ x)

theorem falsa_152_1867 : ∀ (G : Type) [Magma G], LeyA G → LeyB G
